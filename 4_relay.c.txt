#include <xc.h>

#define _XTAL_FREQ 20000000

// Configuration bits
#pragma config FOSC = HS
#pragma config WDT = OFF
#pragma config LVP = OFF
#pragma config PBADEN = OFF

void main(void)
{
    // Configure analog-capable pins as digital
    ADCON1 = 0x0F;

    // RA3 as input
    TRISAbits.TRISA3 = 1;

    // RC1 as output
    TRISCbits.TRISC1 = 0;

    // Initially relay OFF
    LATCbits.LATC1 = 0;

    while(1)
    {
        if(PORTAbits.RA3 == 1)
        {
            LATCbits.LATC1 = 1;   // Relay ON
        }
        else
        {
            LATCbits.LATC1 = 0;   // Relay OFF
        }
    }
}
