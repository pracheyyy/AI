#include <p18f4550.h>
#pragma config FOSC = HS
#pragma config WDT = OFF
#pragma config LVP = OFF
#pragma config PBADEN = OFF
void main(void)
{
    ADCON1 = 0x0F;          // Configure all analog pins as digital

    TRISBbits.TRISB0 = 1;   // RB0 = INPUT (Switch)
    TRISDbits.TRISD0 = 0;   // RD0 = OUTPUT (LED)

    LATDbits.LATD0 = 0;     // Initially LED OFF

    while(1)
    {
        if(PORTBbits.RB0 == 1)   // Switch ON
        {
            LATDbits.LATD0 = 1;  // LED ON
        }
        else                     // Switch OFF
        {
            LATDbits.LATD0 = 0;  // LED OFF
        }
    }
}
